rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {
    function signedIn() {
      return request.auth != null;
    }

    function doctorProfileExists() {
      return signedIn()
        && exists(/databases/$(database)/documents/doctors/$(request.auth.uid))
        && get(/databases/$(database)/documents/doctors/$(request.auth.uid)).data.role == 'doctor';
    }

    function adminProfileExists() {
      return signedIn()
        && exists(/databases/$(database)/documents/admins/$(request.auth.uid))
        && get(/databases/$(database)/documents/admins/$(request.auth.uid)).data.role == 'admin';
    }

    function roomProfileExists() {
      return signedIn()
        && exists(/databases/$(database)/documents/rooms/$(request.auth.uid))
        && get(/databases/$(database)/documents/rooms/$(request.auth.uid)).data.role == 'room';
    }

    function isClinicalStaff() {
      return doctorProfileExists() || adminProfileExists() || roomProfileExists();
    }

    function canAssignDiagnosticTests() {
      return doctorProfileExists() || adminProfileExists();
    }

    function ownsPatient(patientId) {
      return signedIn()
        && (
          request.auth.uid == patientId
          || (
            exists(/databases/$(database)/documents/patients/$(patientId))
            && get(/databases/$(database)/documents/patients/$(patientId)).data.email
              == request.auth.token.email
          )
        );
    }

    function ownsQueueEntry() {
      return signedIn()
        && (
          resource.data.patientId == request.auth.uid
          || resource.data.patientEmail == request.auth.token.email
        );
    }

    match /patients/{patientId} {
      allow get: if ownsPatient(patientId) || isClinicalStaff();
      allow list: if isClinicalStaff()
        || (signedIn() && resource.data.email == request.auth.token.email);

      allow create: if signedIn()
        && (
          (
            request.auth.uid == patientId
            && request.resource.data.uid == request.auth.uid
          )
          || (
            adminProfileExists()
            && request.resource.data.accountNo is string
            && request.resource.data.fullName is string
          )
        );

      allow update: if ownsPatient(patientId)
        && request.resource.data.diff(resource.data).affectedKeys()
          .hasOnly([
            'sex',
            'age',
            'birthday',
            'contact',
            'address',
            'profileCompleted',
            'updatedAt'
          ])
        || (
          (doctorProfileExists() || roomProfileExists())
          && request.resource.data.diff(resource.data).affectedKeys()
            .hasOnly(['allergies', 'updatedAt'])
        );

      allow delete: if adminProfileExists();

      match /medicalRecords/{recordId} {
        allow read: if ownsPatient(patientId) || isClinicalStaff();
        allow create: if (doctorProfileExists() || roomProfileExists())
          && request.resource.data.doctorId == request.auth.uid;
        allow update: if (doctorProfileExists() || roomProfileExists())
          && resource.data.doctorId == request.auth.uid
          && request.resource.data.doctorId == request.auth.uid;
        allow delete: if (doctorProfileExists() || roomProfileExists())
          && resource.data.doctorId == request.auth.uid;
      }

      match /diagnosticPlans/{planId} {
        allow read: if ownsPatient(patientId) || isClinicalStaff();
        allow create: if canAssignDiagnosticTests()
          && request.resource.data.keys().hasOnly([
            'diagnosis',
            'assignedBy',
            'testCount',
            'createdAt'
          ])
          && request.resource.data.diagnosis is string
          && request.resource.data.diagnosis.size() > 0
          && request.resource.data.assignedBy is string
          && request.resource.data.testCount is int
          && request.resource.data.testCount > 0;
        allow update, delete: if false;
      }

      match /diagnosticTests/{testId} {
        allow read: if ownsPatient(patientId) || isClinicalStaff();
        allow create: if canAssignDiagnosticTests()
          && request.resource.data.keys().hasOnly([
            'planId',
            'category',
            'name',
            'description',
            'diagnosis',
            'assignedBy',
            'status',
            'assignedAt',
            'updatedAt'
          ])
          && request.resource.data.planId is string
          && request.resource.data.category is string
          && request.resource.data.name is string
          && request.resource.data.description is string
          && request.resource.data.diagnosis is string
          && request.resource.data.assignedBy is string
          && request.resource.data.status == 'ordered';
        allow update: if isClinicalStaff()
          && request.resource.data.diff(resource.data).affectedKeys()
            .hasOnly(['status', 'updatedAt'])
          && request.resource.data.status in ['ordered', 'in_progress', 'completed'];
        allow delete: if false;
      }

      match /emergencyProcedureProgress/{routingId} {
        allow read, create, update: if ownsPatient(patientId) || isClinicalStaff();
        allow delete: if false;
      }
    }

    match /appointments/{appointmentId} {
      allow get, list: if isClinicalStaff()
        || (signedIn() && resource.data.patientId == request.auth.uid);

      allow create: if signedIn()
        && request.resource.data.patientId == request.auth.uid
        && request.resource.data.slotId
          == request.resource.data.date + '_' + request.resource.data.timeSlot
        && existsAfter(
          /databases/$(database)/documents/appointmentSlots/$(request.resource.data.slotId)
        )
        && getAfter(
          /databases/$(database)/documents/appointmentSlots/$(request.resource.data.slotId)
        ).data.appointmentId == appointmentId
        && request.resource.data.keys().hasOnly([
          'patientId',
          'slotId',
          'accountNo',
          'patientName',
          'date',
          'timeSlot',
          'serviceType',
          'status',
          'createdAt'
        ])
        && request.resource.data.status == 'pending'
        && request.resource.data.accountNo is string
        && request.resource.data.patientName is string
        && request.resource.data.date is string
        && request.resource.data.timeSlot is string
        && request.resource.data.serviceType is string;

      allow update: if isClinicalStaff()
        || (
          signedIn()
          && resource.data.patientId == request.auth.uid
          && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['status'])
          && request.resource.data.status == 'cancelled'
        );
      allow delete: if adminProfileExists();
    }

    match /appointmentSlots/{slotId} {
      allow get, list: if signedIn();
      allow create: if signedIn()
        && request.resource.data.keys().hasOnly(['date', 'timeSlot', 'appointmentId'])
        && slotId == request.resource.data.date + '_' + request.resource.data.timeSlot
        && request.resource.data.date is string
        && request.resource.data.timeSlot is string
        && request.resource.data.appointmentId is string
        && existsAfter(
          /databases/$(database)/documents/appointments/$(request.resource.data.appointmentId)
        )
        && getAfter(
          /databases/$(database)/documents/appointments/$(request.resource.data.appointmentId)
        ).data.patientId == request.auth.uid
        && getAfter(
          /databases/$(database)/documents/appointments/$(request.resource.data.appointmentId)
        ).data.slotId == slotId
        && getAfter(
          /databases/$(database)/documents/appointments/$(request.resource.data.appointmentId)
        ).data.date == request.resource.data.date
        && getAfter(
          /databases/$(database)/documents/appointments/$(request.resource.data.appointmentId)
        ).data.timeSlot == request.resource.data.timeSlot;
      allow update: if false;
      allow delete: if adminProfileExists()
        || (
          signedIn()
          && exists(/databases/$(database)/documents/appointments/$(resource.data.appointmentId))
          && get(
            /databases/$(database)/documents/appointments/$(resource.data.appointmentId)
          ).data.patientId == request.auth.uid
        );
    }

    match /{path=**}/appointments/{appointmentId} {
      allow read: if isClinicalStaff()
        || (signedIn() && resource.data.patientId == request.auth.uid);
      allow update, delete: if isClinicalStaff();
      allow create: if false;
    }

    // This public projection contains only patient ID and queue priority, never names or clinical details.
    match /queuePositions/{queueId} {
      allow get, list: if signedIn();
      allow create, update: if isClinicalStaff()
        && request.resource.data.keys().hasOnly(['accountNo', 'priorityWeight'])
        && request.resource.data.keys().hasAll(['accountNo', 'priorityWeight'])
        && request.resource.data.accountNo == queueId
        && request.resource.data.priorityWeight is number;
      allow delete: if isClinicalStaff();
    }

    match /queues/{queueId} {
      allow get, list: if isClinicalStaff() || ownsQueueEntry();
      allow create, update, delete: if isClinicalStaff();
    }

    match /prescriptions/{prescriptionId} {
      allow read: if isClinicalStaff()
        || (
          signedIn()
          && exists(/databases/$(database)/documents/patients/$(request.auth.uid))
          && resource.data.userId
            == get(/databases/$(database)/documents/patients/$(request.auth.uid)).data.accountNo
        );
      allow create: if (doctorProfileExists() || roomProfileExists())
        && request.resource.data.userId is string
        && request.resource.data.doctorName is string
        && request.resource.data.prescriptionText is string;
      allow update, delete: if adminProfileExists();
    }

    match /doctors/{doctorId} {
      allow get: if signedIn()
        && (request.auth.uid == doctorId || isClinicalStaff());
      allow list: if isClinicalStaff()
        || (signedIn() && resource.data.email == request.auth.token.email);
      allow create, update, delete: if adminProfileExists();
    }

    match /admins/{adminId} {
      allow get: if signedIn() && request.auth.uid == adminId;
      allow list: if signedIn() && resource.data.email == request.auth.token.email;
      allow create, update, delete: if false;
    }

    match /rooms/{roomId} {
      allow get: if signedIn()
        && (request.auth.uid == roomId || adminProfileExists());
      allow list: if adminProfileExists();
      allow create: if adminProfileExists()
        && request.resource.data.uid == roomId
        && request.resource.data.role == 'room'
        && request.resource.data.roomName is string
        && request.resource.data.email is string;
      allow update, delete: if false;
    }

    match /emergencyProtocols/{routingId} {
      allow read: if signedIn();
      allow create, update: if adminProfileExists()
        && request.resource.data.keys().hasOnly([
          'routingId',
          'emergencyType',
          'steps',
          'version',
          'updatedBy',
          'updatedAt'
        ])
        && request.resource.data.routingId == routingId
        && request.resource.data.emergencyType is string
        && request.resource.data.steps is list
        && request.resource.data.steps.size() > 0
        && request.resource.data.version is string
        && request.resource.data.updatedBy == request.auth.uid;
      allow delete: if adminProfileExists();
    }

    match /er_routing/{routingId} {
      allow get, list: if signedIn();
      allow create, update, delete: if adminProfileExists();
    }

    match /clinic_mappings/{clinicId} {
      allow get, list: if signedIn();
      allow create, update, delete: if adminProfileExists();
    }

    match /counters/{counterId} {
      allow read, write: if adminProfileExists();
    }

    match /queue/{queueId} {
      allow read: if isClinicalStaff();
      allow write: if adminProfileExists();
    }

    match /users/{userId} {
      allow read: if isClinicalStaff();
      allow write: if false;
    }

    match /hospitals/{hospitalId} {
      allow read: if signedIn();
      allow write: if adminProfileExists();

      match /er_routing/{routingId} {
        allow read: if signedIn();
        allow write: if adminProfileExists();
      }
    }
  }
}
